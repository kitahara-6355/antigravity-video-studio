## `netguard.py` — 外向き遮断（loopback だけ通す）

```python
# -*- coding: utf-8 -*-
"""pytest の外で TestClient を回すとき用の外向き遮断。**loopback だけ通す**。

全部潰すと Windows の `asyncio` が `socketpair()` で落ちて、
**全リクエストが例外なのに「無印0件」と表示される**（8周目の罠）。
"""
import socket as _s

_orig_connect = _s.socket.connect
_orig_connect_ex = _s.socket.connect_ex


def _is_loopback(addr):
    try:
        host = addr[0] if isinstance(addr, (tuple, list)) else None
    except Exception:
        return False
    if host is None:
        return True  # AF_UNIX 等はそのまま通す
    return host in ("127.0.0.1", "::1", "localhost", "0.0.0.0", "")


def _guard(self, addr):
    if not _is_loopback(addr):
        raise OSError(f"[netguard] outbound blocked: {addr}")
    return _orig_connect(self, addr)


def _guard_ex(self, addr):
    if not _is_loopback(addr):
        return 1
    return _orig_connect_ex(self, addr)


_s.socket.connect = _guard
_s.socket.connect_ex = _guard_ex

```

## `sweep_silent_defaults.py` — 式単位の掃引（黙って定数に落ちる読み口）

```python
# -*- coding: utf-8 -*-
"""面(b) を**鍵単位**で掃引する — 「黙って定数に落ちる読み口」を全部出す。

16・17周目の反例はどちらも**同じ形**だった:

    ranks?.biz_rank?.xp || 10        ← `ranks` が存在せず、常に 10
    statusData.overall_score ?? …    ← 定数 85 を実測の顔で

13・15周目も同型（`|| 0` / 直書き）。つまり C4 のフロント側の反例は
**「データ経路 → 既定のリテラル」という1つのクラス**に収まる。

ファイル単位の掃引（16周目まで）ではこれを取り逃した。
**式単位**で、`?.` を含む読み口が数値・文字列リテラルへ落ちる箇所を全部出す。
"""
import json
import re
from pathlib import Path

ROOT = Path(r"C:\Users\PC_User\Desktop\script\antigravity-video-studio")
SRC = ROOT / "frontend" / "src"

# `a?.b?.c || 123` / `a?.b ?? 'text'` を拾う。
# 左辺に `?.` があり（＝API 由来の可能性が高い）、右辺がリテラル
式 = re.compile(
    r"([A-Za-z_$][\w$]*(?:\?\.[\w$]+|\.[\w$]+|\?\.\[[^\]]+\])+)"   # 読み口
    r"\s*(\|\||\?\?)\s*"                                            # || か ??
    r"(-?\d+(?:\.\d+)?|'[^']{1,60}'|\"[^\"]{1,60}\")"               # リテラル
)

# 収益化・品質の判断に関わる語（これを含む読み口は要注意）
要注意 = ("score", "Score", "xp", "XP", "rank", "Rank", "subs", "subscriber",
        "views", "ctr", "CTR", "retention", "維持", "quality", "品質",
        "count", "total", "rate", "率", "achievement", "達成")

見つかった = []
for p in sorted(SRC.rglob("*.jsx")) + sorted(SRC.rglob("*.js")):
    if "node_modules" in p.parts:
        continue
    src = p.read_text(encoding="utf-8", errors="ignore")
    # コメントを外す（説明文の中の旧コードを拾わない）
    本文 = re.sub(r"/\*.*?\*/", "", src, flags=re.S)
    本文 = "\n".join(行 for 行 in 本文.splitlines()
                   if not 行.strip().startswith("//"))
    for m in 式.finditer(本文):
        読み口, 演算子, リテラル = m.group(1), m.group(2), m.group(3)
        # 数値 0 / 空文字 / 短い定型文言への fallback は fail-closed 側なので除く
        if リテラル in ("0", "'0'", '"0"', "''", '""'):
            continue
        危険 = any(k in 読み口 for k in 要注意)
        行番号 = 本文[:m.start()].count("\n") + 1
        見つかった.append({
            "file": p.relative_to(ROOT).as_posix(),
            "line": 行番号,
            "expr": f"{読み口} {演算子} {リテラル}",
            "要注意": 危険,
        })

危険 = [x for x in 見つかった if x["要注意"]]
print(f"# `?.` の読み口がリテラルに落ちる箇所: {len(見つかった)} 件")
print(f"## うち 品質・統計に関わる語を含むもン（要点検）: {len(危険)} 件")
print()
for x in 危険:
    print(f"  {x['file']}:{x['line']}")
    print(f"      {x['expr']}")

出力 = Path(r"C:\Users\PC_User\AppData\Local\Temp\claude\C--Users-PC-User-Desktop-script-antigravity-video-studio\673fb77e-80ad-4883-a0b3-fc49e4dd5797\scratchpad\silent_defaults.json")
出力.write_text(json.dumps(見つかった, ensure_ascii=False, indent=1), encoding="utf-8")
print(f"\n# 全件: {出力}")
```
